cmake_minimum_required(VERSION 3.25)
project(VoltageGarden VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
 add_compile_options(/utf-8)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
option(GARDEN_PLUGIN "Build JUCE VST3 and standalone" ON)
include(CTest)
add_library(garden_core INTERFACE)
target_include_directories(garden_core INTERFACE src)
add_executable(garden_tests tests/CoreTests.cpp)
target_link_libraries(garden_tests PRIVATE garden_core)
if(MSVC)
 target_compile_options(garden_tests PRIVATE /W4 /permissive-)
else()
 target_compile_options(garden_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME core COMMAND garden_tests)
if(GARDEN_PLUGIN)
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850)
 FetchContent_MakeAvailable(JUCE)
 juce_add_plugin(VoltageGarden COMPANY_NAME "Jerzy Audio" BUNDLE_ID "pl.jerzyaudio.voltagegarden" PLUGIN_MANUFACTURER_CODE Jrzy PLUGIN_CODE Vgdn FORMATS VST3 Standalone PRODUCT_NAME "Voltage Garden" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
 target_sources(VoltageGarden PRIVATE src/Plugin.cpp src/Editor.cpp)
 target_link_libraries(VoltageGarden PRIVATE garden_core juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
 target_compile_definitions(VoltageGarden PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 juce_add_console_app(garden_integration PRODUCT_NAME "Garden integration")
 target_sources(garden_integration PRIVATE tests/PluginTests.cpp src/Plugin.cpp src/Editor.cpp)
 target_include_directories(garden_integration PRIVATE src)
 target_link_libraries(garden_integration PRIVATE garden_core juce::juce_audio_utils juce::juce_dsp)
 target_compile_definitions(garden_integration PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JucePlugin_Name="VoltageGarden")
 add_test(NAME plugin_integration COMMAND garden_integration)
endif()
